39def _read_assignments( 

40 path: Path, expected_ids: set[str], foldseek_models: bool = False 

41) -> pd.DataFrame: 

42 """Require one native-tool assignment per submitted input.""" 

43 assignments = pd.read_csv( 

44 path, 

45 sep="\t", 

46 header=None, 

47 names=["representative", "member"], 

48 dtype=str, 

49 keep_default_na=False, 

50 ) 

51 if foldseek_models: 

52 for column in ("representative", "member"): 

53 original = assignments[column] 

54 without_model = original.str.replace( 

55 r"_xyz-enrich_MODEL_[0-9]+_", "_xyz-enrich_", n=1, regex=True 

56 ) 

57 assignments[column] = original.mask( 

58 ~original.isin(expected_ids) & without_model.isin(expected_ids), 

59 without_model, 

60 ) 

61 observed = set(assignments["member"]) 

62 if len(assignments) != len(expected_ids) or observed != expected_ids: 

63 raise ValueError( 

64 "protein cluster membership differs from input: " 

65 f"rows={len(assignments)}, expected={len(expected_ids)}, " 

66 f"missing={sorted(expected_ids - observed)[:10]}, " 

67 f"extra={sorted(observed - expected_ids)[:10]}" 

68 ) 

69 return assignments

72def _write_cluster_table( 

73 chains: pd.DataFrame, 

74 assignments: pd.DataFrame, 

75 output: Path, 

76 metadata: bytes, 

77 *, 

78 missing_status: str | dict[str, str], 

79) -> None: 

80 representative_keys = chains[["member", "entry_pdb_id", "chain_asym_id"]].rename( 

81 columns={ 

82 "member": "representative", 

83 "entry_pdb_id": "representative_entry_pdb_id", 

84 "chain_asym_id": "representative_chain_asym_id", 

85 } 

86 ) 

87 result = chains.merge(assignments, on="member", how="left", validate="one_to_one") 

88 result = result.merge( 

89 representative_keys, on="representative", how="left", validate="many_to_one" 

90 ) 

91 result["is_representative"] = ( 

92 result["member"].eq(result["representative"]).astype("boolean") 

93 ) 

94 missing = result["representative"].isna() 

95 result.loc[missing, "is_representative"] = pd.NA 

96 result["status"] = "clustered" 

97 result.loc[missing, "status"] = ( 

98 result.loc[missing, "member"].map(missing_status) 

99 if isinstance(missing_status, dict) 

100 else missing_status 

101 ) 

102 table = pa.Table.from_pandas( 

103 result[SEQUENCE_CLUSTER_SCHEMA.names], 

104 schema=SEQUENCE_CLUSTER_SCHEMA, 

105 preserve_index=False, 

106 ).replace_schema_metadata({CLUSTER_METADATA_KEY: metadata}) 

107 output.parent.mkdir(parents=True, exist_ok=True) 

108 # Stage beside the destination so replacement is atomic across filesystems. 

109 with TemporaryDirectory(prefix=f".{output.stem}-", dir=output.parent) as staging: 

110 staged = Path(staging) / output.name 

111 pq.write_table(table, staged, compression="zstd") 

112 staged.replace(output)

115def make_protein_structure_clusters( 

116 *, 

117 data_dir: Path, 

118 scratch_dir: Path, 

119 threads: int = 4, 

120 lddt: float = 0.7, 

121 coverage: float = 0.8, 

122 force_update: bool = False, 

123) -> Path: 

124 """Cluster release protein chains from the existing Foldseek search database. 

125 

126 The search database contains all chains parsed from the deposited mmCIFs. 

127 A temporary subdatabase selects the release protein chains, avoiding a 

128 second read of every source structure and per-chain mmCIF exports. 

129 """ 

130 

131 if threads < 1: 131 ↛ 132line 131 didn't jump to line 132 because the condition on line 131 was never true

132 raise ValueError("protein clustering threads must be positive") 

133 if not 0 <= lddt <= 1 or not 0 <= coverage <= 1: 133 ↛ 134line 133 didn't jump to line 134 because the condition on line 133 was never true

134 raise ValueError("protein clustering lDDT and coverage must be between 0 and 1") 

135 chains = ( 

136 pd.read_parquet( 

137 data_dir / "index/entry_chains.parquet", 

138 columns=["entry_pdb_id", "chain_asym_id", "chain_auth_id"], 

139 filters=[("chain_receptor_type", "==", "protein")], 

140 ) 

141 .sort_values(["entry_pdb_id", "chain_asym_id"]) 

142 .reset_index(drop=True) 

143 ) 

144 if chains["chain_auth_id"].isna().any(): 144 ↛ 145line 144 didn't jump to line 145 because the condition on line 144 was never true

145 raise ValueError("protein chains require author chain IDs for Foldseek lookup") 

146 chains["member"] = ( 

147 "pdb_0000" 

148 + chains["entry_pdb_id"].astype(str) 

149 + "_xyz-enrich_" 

150 + chains["chain_auth_id"].astype(str) 

151 ) 

152 if chains["member"].duplicated().any(): 

153 raise ValueError("protein chains have ambiguous Foldseek author chain IDs") 

154 db = data_dir / "dbs/foldseek/foldseek" 

155 lookup = Path(f"{db}.lookup") 

156 source_manifest = Path(f"{db}.createdb-input.json") 

157 for required in (db, lookup, source_manifest): 

158 if not required.is_file(): 

159 raise FileNotFoundError(required) 

160 digest = hashlib.sha256() 

161 digest.update( 

162 chains[["entry_pdb_id", "chain_asym_id", "member"]] 

163 .to_json(orient="values") 

164 .encode() 

165 ) 

166 digest.update(source_manifest.read_bytes()) 

167 for path in (db, lookup): 

168 stat = path.stat() 

169 digest.update(f"{stat.st_size}:{stat.st_mtime_ns}".encode()) 

170 parameters = { 

171 "backend": "foldseek", 

172 "version": subprocess.check_output(["foldseek", "version"], text=True).strip(), 

173 "lddt": lddt, 

174 "coverage": coverage, 

175 "coverage_mode": 0, 

176 "cluster_mode": 0, 

177 "single_step_clustering": True, 

178 "alignment_type": 2, 

179 "source_database_signature": digest.hexdigest(), 

180 } 

181 metadata = json.dumps(parameters, sort_keys=True).encode() 

182 output = data_dir / RELEASE_PATHS["protein_structure_clusters"] 

183 if not force_update and _clusters_are_current(output, metadata): 

184 return output 

185 scratch_dir.mkdir(parents=True, exist_ok=True) 

186 with TemporaryDirectory( 

187 prefix="plinder-structure-clusters-", dir=scratch_dir 

188 ) as temporary: 

189 work = Path(temporary) 

190 expected = set(chains["member"]) 

191 keys: dict[str, str] = {} 

192 with lookup.open() as handle: 

193 for line in handle: 

194 key, name, _ = line.rstrip("\n").split("\t", 2) 

195 if name in expected and name not in keys: 

196 # For NMR entries this is the first available model. 

197 keys[name] = key 

198 if keys: 

199 key_file = work / "keys.tsv" 

200 key_file.write_text("".join(f"{key}\n" for key in keys.values())) 

201 subdb = work / "subdb" 

202 run(["foldseek", "createsubdb", str(key_file), str(db), str(subdb)]) 

203 run( 

204 [ 

205 "foldseek", 

206 "cluster", 

207 str(subdb), 

208 str(work / "clusters"), 

209 str(work / "tmp"), 

210 "--lddt-threshold", 

211 str(lddt), 

212 "-c", 

213 str(coverage), 

214 "--cov-mode", 

215 "0", 

216 "--cluster-mode", 

217 "0", 

218 "--single-step-clustering", 

219 "1", 

220 "--alignment-type", 

221 "2", 

222 "--threads", 

223 str(threads), 

224 ] 

225 ) 

226 assignment_file = work / "clusters.tsv" 

227 run( 

228 [ 

229 "foldseek", 

230 "createtsv", 

231 str(subdb), 

232 str(subdb), 

233 str(work / "clusters"), 

234 str(assignment_file), 

235 "--threads", 

236 str(threads), 

237 ] 

238 ) 

239 assignments = _read_assignments( 

240 assignment_file, set(keys), foldseek_models=True 

241 ) 

242 else: 

243 assignments = pd.DataFrame(columns=["representative", "member"]) 

244 _write_cluster_table( 

245 chains, 

246 assignments, 

247 output, 

248 metadata, 

249 missing_status="not_in_foldseek_db", 

250 ) 

251 return output

254def make_protein_sequence_clusters( 

255 *, 

256 data_dir: Path, 

257 scratch_dir: Path, 

258 threads: int = 4, 

259 identity: float = 0.4, 

260 coverage: float = 0.8, 

261 force_update: bool = False, 

262) -> Path: 

263 """Cluster all release protein-chain sequences using MMseqs easy-cluster. 

264 

265 Each representative/member match must meet the identity threshold and cover 

266 the requested fraction of both chains. Greedy set cover runs in single-step 

267 mode. Apo and interface chains are included. Missing sequences have null 

268 assignments and ``status='missing_sequence'`` in the output. 

269 

270 Intermediate MMseqs files live under ``scratch_dir`` and are removed after 

271 the table is written. Parameters, input sequence digest, and MMseqs version 

272 are stored in Parquet metadata to check whether a previous result is reusable. 

273 """ 

274 if threads < 1: 274 ↛ 275line 274 didn't jump to line 275 because the condition on line 274 was never true

275 raise ValueError("protein clustering threads must be positive") 

276 if not 0 <= identity <= 1 or not 0 <= coverage <= 1: 276 ↛ 277line 276 didn't jump to line 277 because the condition on line 276 was never true

277 raise ValueError( 

278 "protein clustering identity and coverage must be between 0 and 1" 

279 ) 

280 chains = ( 

281 pd.read_parquet( 

282 data_dir / "index" / "entry_chains.parquet", 

283 columns=["entry_pdb_id", "chain_asym_id", "chain_sequence"], 

284 filters=[("chain_receptor_type", "==", "protein")], 

285 ) 

286 .sort_values(["entry_pdb_id", "chain_asym_id"]) 

287 .reset_index(drop=True) 

288 ) 

289 keys = ["entry_pdb_id", "chain_asym_id"] 

290 # Generated FASTA IDs avoid any restrictions in native tools on source IDs. 

291 # They are mapped back to the untouched entry/asym IDs before writing. 

292 chains["member"] = pd.Series( 

293 [f"chain_{i}" for i in range(len(chains))], dtype="string" 

294 ) 

295 valid = chains["chain_sequence"].notna() & chains["chain_sequence"].ne("") 

296 sequences = chains.loc[valid] 

297 if not sequences["chain_sequence"].str.fullmatch("[A-Za-z]+").all(): 297 ↛ 298line 297 didn't jump to line 298 because the condition on line 297 was never true

298 raise ValueError( 

299 "protein chain sequences must contain one-letter amino-acid codes" 

300 ) 

301 digest = hashlib.sha256() 

302 for row in chains[keys + ["chain_sequence"]].itertuples(index=False, name=None): 

303 digest.update( 

304 json.dumps([None if pd.isna(value) else value for value in row]).encode() 

305 ) 

306 digest.update(b"\n") 

307 parameters = { 

308 "backend": "mmseqs", 

309 "sequence_type": "protein", 

310 "version": subprocess.check_output(["mmseqs", "version"], text=True).strip(), 

311 "identity": identity, 

312 "coverage": coverage, 

313 "coverage_mode": 0, 

314 "cluster_mode": 0, 

315 "single_step_clustering": True, 

316 "sequence_digest": digest.hexdigest(), 

317 } 

318 metadata = json.dumps(parameters, sort_keys=True).encode() 

319 output = data_dir / RELEASE_PATHS["protein_sequence_clusters"] 

320 if not force_update and _clusters_are_current(output, metadata): 

321 return output 

322 scratch_dir.mkdir(parents=True, exist_ok=True) 

323 with TemporaryDirectory( 

324 prefix="plinder-sequence-clusters-", dir=scratch_dir 

325 ) as temporary: 

326 work = Path(temporary) 

327 if sequences.empty: 

328 assignments = pd.DataFrame(columns=["representative", "member"]) 

329 else: 

330 fasta = work / "chains.fasta" 

331 with fasta.open("w") as handle: 

332 for row in sequences.itertuples(index=False): 

333 handle.write(f">{row.member}\n{row.chain_sequence}\n") 

334 run( 

335 [ 

336 "mmseqs", 

337 "easy-cluster", 

338 str(fasta), 

339 str(work / "clusters"), 

340 str(work / "tmp"), 

341 "--dbtype", 

342 "1", 

343 "--min-seq-id", 

344 str(identity), 

345 "-c", 

346 str(coverage), 

347 "--cov-mode", 

348 "0", 

349 "--cluster-mode", 

350 "0", 

351 "--single-step-clustering", 

352 "1", 

353 "--threads", 

354 str(threads), 

355 ] 

356 ) 

357 assignments = _read_assignments( 

358 work / "clusters_cluster.tsv", set(sequences["member"]) 

359 ) 

360 _write_cluster_table( 

361 chains, assignments, output, metadata, missing_status="missing_sequence" 

362 ) 

363 return output
