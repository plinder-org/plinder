228def _predict_prodigy_probability(features: np.ndarray) -> tuple[str, float]: 

229 """Evaluate the exact trees from official PRODIGY-cryst.""" 

230 model = _load_prodigy_classifier() 

231 offsets = model["tree_offsets"] 

232 left = model["children_left"] 

233 right = model["children_right"] 

234 node_features = model["feature"] 

235 thresholds = model["threshold"] 

236 leaf_probabilities = model["probability"] 

237 probabilities = np.zeros(2, dtype=np.float64) 

238 for tree_index in range(len(offsets) - 1): 

239 node = int(offsets[tree_index]) 

240 while left[node] >= 0: 

241 if features[node_features[node]] <= thresholds[node]: 

242 node = int(left[node]) 

243 else: 

244 node = int(right[node]) 

245 probabilities += leaf_probabilities[node] 

246 probabilities /= len(offsets) - 1 

247 return ("BIO" if probabilities[0] >= probabilities[1] else "XTAL"), float( 

248 probabilities[0]

252def annotate_prodigy_crystal_interface( 

253 atoms: struc.AtomArray, 

254 *, 

255 chain_1: str, 

256 chain_2: str, 

257 spatial_index: BiounitSpatialIndex, 

258) -> ProdigyCrystalAnnotation | None: 

259 """Classify one chain interface using official PRODIGY-cryst. 

260 

261 Its contact definitions, feature ordering and fixed random forest are 

262 preserved. Biotite supplies the already reconstructed biological 

263 assembly, and NumPy evaluates the legacy scikit-learn trees. 

264 Interfaces containing unsupported modified residues remain in the dataset 

265 with null PRODIGY fields. 

266 """ 

267 chain_1_mask = atoms.chain_id == chain_1 

268 chain_2_mask = atoms.chain_id == chain_2 

269 if not np.any(chain_1_mask) or not np.any(chain_2_mask): 269 ↛ 270line 269 didn't jump to line 270 because the condition on line 269 was never true

270 raise ValueError( 

271 f"interface chains are absent from assembly: {chain_1}, {chain_2}" 

272 ) 

273 chain_2_atom_mask = np.asarray(chain_2_mask, dtype=bool) 

274 contact_pairs: set[tuple[tuple[int, str], tuple[int, str]]] = set() 

275 chain_1_indices = np.flatnonzero(chain_1_mask) 

276 for atom_index in chain_1_indices: 

277 neighbors = np.asarray( 

278 spatial_index.cell_list.get_atoms( 

279 atoms.coord[atom_index], radius=PRODIGY_CONTACT_RADIUS 

280 ), 

281 dtype=int, 

282 ).reshape(-1) 

283 neighbors = neighbors[(neighbors >= 0) & (neighbors < len(atoms))] 

284 neighbors = neighbors[chain_2_atom_mask[neighbors]] 

285 residue_1 = (int(atoms.res_id[atom_index]), str(atoms.ins_code[atom_index])) 

286 for target_index in neighbors: 

287 residue_2 = ( 

288 int(atoms.res_id[target_index]), 

289 str(atoms.ins_code[target_index]), 

290 ) 

291 contact_pairs.add((residue_1, residue_2)) 

292 if not contact_pairs: 

293 return None 

294 

295 residue_name_1 = { 

296 (int(atoms.res_id[index]), str(atoms.ins_code[index])): str( 

297 atoms.res_name[index] 

298 ) 

299 for index in np.flatnonzero(chain_1_mask) 

300 } 

301 residue_name_2 = { 

302 (int(atoms.res_id[index]), str(atoms.ins_code[index])): str( 

303 atoms.res_name[index] 

304 ) 

305 for index in np.flatnonzero(chain_2_mask) 

306 } 

307 contacted_residue_names = { 

308 residue_name_1[residue_1] for residue_1, _ in contact_pairs 

309 } | {residue_name_2[residue_2] for _, residue_2 in contact_pairs} 

310 if not contacted_residue_names.issubset(PRODIGY_AMINO_ACID_CLASSES): 

311 return None 

312 bins = {name: 0 for name in ("AA", "PP", "CC", "AP", "CP", "AC")} 

313 bins.update({name: 0 for name in PRODIGY_AMINO_ACID_CLASSES}) 

314 for residue_1, residue_2 in contact_pairs: 

315 name_1 = residue_name_1[residue_1] 

316 name_2 = residue_name_2[residue_2] 

317 contact_type = "".join( 

318 sorted( 

319 ( 

320 PRODIGY_AMINO_ACID_CLASSES[name_1], 

321 PRODIGY_AMINO_ACID_CLASSES[name_2], 

322 ) 

323 ) 

324 ) 

325 bins[contact_type] += 1 

326 bins[name_1] += 1 

327 bins[name_2] += 1 

328 contacted_1 = {pair[0] for pair in contact_pairs} 

329 contacted_2 = {pair[1] for pair in contact_pairs} 

330 link_density = len(contact_pairs) / (len(contacted_1) * len(contacted_2)) 

331 features = np.asarray( 

332 [float(bins[name]) for name in PRODIGY_FEATURE_NAMES[:-1]] + [link_density], 

333 dtype=np.float64, 

334 ) 

335 label, probability_bio = _predict_prodigy_probability(features) 

336 return ProdigyCrystalAnnotation( 

337 label=label, 

338 probability_bio=probability_bio, 

339 link_density=link_density, 

340 intermolecular_contacts=len(contact_pairs), 

341 charged_charged_contacts=bins["CC"], 

342 charged_polar_contacts=bins["CP"], 

343 charged_apolar_contacts=bins["AC"], 

344 polar_polar_contacts=bins["PP"], 

345 apolar_polar_contacts=bins["AP"], 

346 apolar_apolar_contacts=bins["AA"], 

347 )
