70def mol2mhfp6(mol: Mol | str) -> NDArray[np.uint32]: 

71 """Return the MHFP6 MinHash fingerprint of a molecule or SMILES string. 

72 

73 MHFP6 hashes the set of circular SMILES shingles (radii 1..3) and keeps the 

74 per-permutation minima, so the fraction of matching positions between two 

75 fingerprints estimates the Jaccard similarity of their shingle sets 

76 (Probst & Reymond, J. Cheminform. 2018). Unlike a folded Morgan bit-vector, 

77 the result is a dense vector of ``MHFP6_N_PERMUTATIONS`` uint32 hashes. 

78 """ 

79 if isinstance(mol, str): 

80 mol = mol_from_smiles(mol) 

81 if mol is None: 

82 raise ValueError("cannot fingerprint an invalid molecule") 

83 try: 

84 encoded = _mhfp6_encoder().EncodeMol(mol, radius=MHFP6_RADIUS) 

85 except RuntimeError as exc: 

86 if "bad bond stereo" not in str(exc): 

87 raise 

88 # RDKit can fail while creating fragments of a conjugated alkene even 

89 # with isomeric=False. Clear only double-bond stereo; keep atom chirality. 

90 mol = Chem.Mol(mol) 

91 for bond in mol.GetBonds(): 

92 if bond.GetBondType() == Chem.BondType.DOUBLE: 

93 bond.SetStereo(Chem.BondStereo.STEREONONE) 

94 elif bond.GetBondDir() != Chem.BondDir.NONE: 

95 bond.SetBondDir(Chem.BondDir.NONE) 

96 encoded = _mhfp6_encoder().EncodeMol(mol, radius=MHFP6_RADIUS) 

97 return np.asarray(encoded, dtype=np.uint32)

128def _opposed_centres( 

129 first: Centres, second: Centres, partner: dict[int, int], atoms: set[int] 

130) -> int: 

131 """Count matched tetrahedral centres of opposite handedness under the mapping. 

132 

133 Tet_CW/Tet_CCW is relative to each molecule's own neighbour order, so the 

134 mapping between the two orders is a permutation and the descriptors are 

135 equal only up to its parity. A centre needs three mapped neighbours; the 

136 leftover slot on each side (fourth neighbour or implicit H, which RDKit 

137 orders last) is taken as corresponding. 

138 """ 

139 opposed = 0 

140 for i in atoms & first.keys(): 

141 j = partner.get(i) 

142 if j not in second: 

143 continue 

144 (descriptor, neighbours), (other_descriptor, other) = first[i], second[j] 

145 # i's neighbour order translated into mol2 indices, next to j's own order 

146 mapped = [partner.get(a, -1) for a in neighbours] 

147 common = set(mapped) & set(other) - {-1} 

148 if len(common) < 3: 

149 continue 

150 mapped, other = ( 

151 [a if a in common else -1 for a in ns] for ns in (mapped, other) 

152 ) 

153 order = [other.index(a) for a in mapped] 

154 odd = sum(order[x] > order[y] for x in range(4) for y in range(x + 1, 4)) % 2 

155 opposed += (descriptor == other_descriptor) == bool(odd) 

156 return opposed

196def parity_similarity( 

197 atoms: float, bonds: float, atom_total: int, bond_total: int 

198) -> float: 

199 """Mean of the atom and bond Tanimotos; totals are both molecules' counts summed."""

205def rascal_parity_match( 

206 mol1: Mol, 

207 mol2: Mol, 

208 *, 

209 target: float = 0.3, 

210 timeout: int = 2, 

211 all_best: bool = False, 

212) -> RascalParityMatch: 

213 """Element-exact Rascal MCES with fragments allowed, pruned to a target bond similarity. 

214 

215 Pairs whose bond Tanimoto cannot reach ``target`` are pruned before the 

216 clique search and come back empty. Rascal maximises matched bonds, and the 

217 MCESs of that size differ in their largest fragment, so the bonds come from 

218 one MCES and the fragment from a second run that keeps only the largest 

219 fragment over all of them (``singleLargestFrag``). With ``all_best`` every 

220 fragment is enumerated instead (a further 1.5x) and the one scoring best is 

221 taken; the rest follow, best first and then in a fixed order. 

222 """ 

223 options = rdRascalMCES.RascalOptions() 

224 # tier screens use Johnson similarity; below (2t/(1+t))^2 no pair reaches bond Tanimoto t 

225 options.similarityThreshold = (2 * target / (1 + target)) ** 2 

226 # fewest matched bonds giving bond Tanimoto >= target; Rascal's bound is exclusive 

227 options.minCliqueSize = max( 

228 int(np.ceil(target * (mol1.GetNumBonds() + mol2.GetNumBonds()) / (1 + target))) 

229 - 1, 

230 0, 

231 ) 

232 options.returnEmptyMCES = True 

233 options.completeAromaticRings = False 

234 options.ringMatchesRingOnly = False 

235 options.maxBondMatchPairs = 5000 

236 options.timeout = timeout 

237 options.allBestMCESs = all_best 

238 matches: dict[tuple[tuple[int, int], ...], dict[int, int]] = {} 

239 bonds = 0 

240 for result in rdRascalMCES.FindMCES(mol1, mol2, options): 

241 matched = result.bondMatches() 

242 if not matched: 

243 continue 

244 # largest connected fragment of the matched bonds, in mol1 atom indices 

245 atom_map: dict[int, int] = {} 

246 core = Chem.PathToSubmol(mol1, [i for i, _ in matched], atomMap=atom_map) 

247 original = {new: old for old, new in atom_map.items()} 

248 partner = dict(result.atomMatches()) 

249 fragment = { 

250 original[i]: partner[original[i]] 

251 for i in max(Chem.GetMolFrags(core), key=len) 

252 } 

253 matches.setdefault(tuple(sorted(fragment.items())), partner) 

254 bonds = len(matched) 

255 if not matches: 

256 return RascalParityMatch({}, 0, 0) 

257 if not all_best: 

258 options.singleLargestFrag = True 

259 best = rdRascalMCES.FindMCES(mol1, mol2, options) 

260 if best and best[0].bondMatches(): 260 ↛ 263line 260 didn't jump to line 263 because the condition on line 260 was always true

261 fragment = dict(best[0].atomMatches()) 

262 matches = {tuple(sorted(fragment.items())): fragment} 

263 centres_1, centres_2 = _stereo_centres(mol1), _stereo_centres(mol2) 

264 opposed = { 

265 key: _opposed_centres(centres_1, centres_2, partner, {i for i, _ in key}) 

266 for key, partner in matches.items() 

267 } 

268 ordered = sorted(matches, key=lambda key: (0.5 * opposed[key] - len(key), key)) 

269 return RascalParityMatch( 

270 dict(ordered[0]), 

271 bonds, 

272 opposed[ordered[0]], 

273 tuple(dict(key) for key in ordered[1:]), 

274 )

277def rascal_parity_score( 

278 mol1: Mol, mol2: Mol, *, target: float = 0.3, stereo: bool = True, timeout: int = 2 

279) -> float: 

280 """PARITY-like similarity from one Rascal MCES, in [0, 1]. 

281 

282 Element-exact MCES with fragments allowed, scored as the mean of two 

283 Tanimoto terms: atoms of the largest connected fragment over the atom union, 

284 and all matched bonds over the bond union. The largest fragment carries 

285 residue order (a shuffled peptide splits into one fragment per residue, a 

286 substitution does not); the bonds keep credit for matches beyond a 

287 substituted connecting atom. With ``stereo`` a matched tetrahedral centre of 

288 opposite handedness counts half an atom. 

289 """ 

290 match = rascal_parity_match(mol1, mol2, target=target, timeout=timeout) 

291 return match.score(mol1, mol2, stereo=stereo)
