64def _tautomer_cache_key(molecule: Chem.Mol) -> _TautomerCacheKey: 

65 """Include atom order and PEPPR split-residue state in the cache key.""" 

66 heavy_neighbors_property = peppr_contacts._ORIG_NUM_HEAVY_NEIGHS 

67 atom_signature = tuple( 

68 ( 

69 atom.GetAtomicNum(), 

70 atom.GetFormalCharge(), 

71 int(atom.GetHybridization()), 

72 atom.GetTotalNumHs(), 

73 atom.GetIsAromatic(), 

74 atom.GetIntProp(heavy_neighbors_property) 

75 if atom.HasProp(heavy_neighbors_property) 

76 else None, 

77 ) 

78 for atom in molecule.GetAtoms() 

79 ) 

80 bond_signature = tuple( 

81 sorted( 

82 ( 

83 min(bond.GetBeginAtomIdx(), bond.GetEndAtomIdx()), 

84 max(bond.GetBeginAtomIdx(), bond.GetEndAtomIdx()), 

85 bond.GetBondTypeAsDouble(), 

86 bond.GetIsAromatic(), 

87 bond.GetIsConjugated(),

89 )

145def _bounded_find_resonance_charges(molecule: Chem.Mol) -> _ResonanceResult: 

146 """Preserve exact PEPPR resonance behavior with a rare timeout fallback.""" 

147 is_large_and_charged = molecule.GetNumAtoms() >= _RESONANCE_GUARD_MIN_ATOMS and any( 

148 atom.GetFormalCharge() != 0 for atom in molecule.GetAtoms() 

149 ) 

150 if not is_large_and_charged: 

151 result: _ResonanceResult = _ORIGINAL_FIND_RESONANCE_CHARGES(molecule) 

152 return result 

153 

154 key = _tautomer_cache_key(molecule) 

155 with _RESONANCE_CACHE_LOCK: 

156 cached = _RESONANCE_CACHE.get(key) 

157 if cached is not None: 

158 _RESONANCE_CACHE.move_to_end(key) 

159 return _copy_resonance_result(cached)

161 # Data ingest is Linux-only and fork lets the child reuse the already 

162 # constructed RDKit molecule without serializing the full contact model. 

163 context = mp.get_context("fork") 

164 receive, send = context.Pipe(duplex=False) 

165 process = context.Process( 

166 target=_run_resonance_worker, 

167 args=(molecule, send), 

168 daemon=True, 

169 ) 

170 process.start() 

171 send.close() 

172 status = "timeout" 

173 payload: Any = None 

174 try: 

175 if receive.poll(_RESONANCE_TIMEOUT_SECONDS): 

176 status, payload = receive.recv() 

177 finally: 

178 receive.close() 

179 if process.is_alive(): 179 ↛ 181line 179 didn't jump to line 181 because the condition on line 179 was always true

180 process.terminate() 

181 process.join(timeout=1.0) 

182 if process.is_alive(): 182 ↛ 183line 182 didn't jump to line 183 because the condition on line 182 was never true

183 process.kill() 

184 process.join() 

185 

186 if status == "ok": 

187 result = cast(_ResonanceResult, payload) 

188 elif status == "error": 188 ↛ 189line 188 didn't jump to line 189 because the condition on line 188 was never true

189 raise struc.BadStructureError( 

190 f"PEPPR resonance enumeration failed in worker: {payload}" 

191 ) 

192 else: 

193 result = _deposited_charge_fallback(molecule) 

194 log.warning( 

195 "PEPPR resonance enumeration exceeded %.1fs for a charged " 

196 "%d-atom residue; using its estimated deposited formal charges", 

197 _RESONANCE_TIMEOUT_SECONDS, 

198 molecule.GetNumAtoms(), 

199 ) 

200 

201 with _RESONANCE_CACHE_LOCK: 

202 _RESONANCE_CACHE[key] = _copy_resonance_result(result) 

203 _RESONANCE_CACHE.move_to_end(key)

216def get_symmetry_mate_contacts( 

217 mmcif: Path | pdbx.CIFFile, contact_threshold: float = 5.0 

218) -> dict[tuple[str, int], dict[tuple[str, int], dict[int, set[int]]]]: 

219 """ 

220 Find inter-residue contacts generated by crystallographic symmetry. 

221 

222 Only contacts involving symmetry mates (not identity) are returned. 

223 

224 Parameters 

225 ---------- 

226 mmcif : Path or CIFFile 

227 Parsed mmCIF or a path to one (supports .gz). Passing a parsed file 

228 avoids reading a large source entry again during ingest. 

229 contact_threshold : float, optional 

230 Distance cutoff in Angstrom, by default 5.0. 

231 

232 Returns 

233 ------- 

234 dict[tuple[str, int], dict[tuple[str, int], dict[int, set[int]]]] 

235 Mapping of (chain_id, residue_id) to partner residues, 

236 with atom serials mapped to the symmetry image indices. 

237 """ 

238 from plinder.data.annotations.cif_utils import ( 

239 get_structure_with_altloc, 

240 get_unit_cell_with_altloc, 

241 read_mmcif_file, 

242 ) 

243 

244 cif_file = mmcif if isinstance(mmcif, pdbx.CIFFile) else read_mmcif_file(mmcif) 

245 

246 # Build the full unit cell (all symmetry copies) 

247 try: 

248 unit_cell = get_unit_cell_with_altloc( 

249 cif_file, model=1, use_author_fields=False 

250 ) 

251 except Exception: 

252 # No symmetry information (NMR, computational models) 

253 return {} 

254 unit_cell = unit_cell[~struc.filter_solvent(unit_cell)] 

255 unit_cell = unit_cell[filter_heavy(unit_cell)] 

256 

257 if unit_cell.box is None: 257 ↛ 258line 257 didn't jump to line 258 because the condition on line 257 was never true

258 return {} 

259 box_lengths = np.linalg.norm(unit_cell.box, axis=1) 

260 if not np.all(np.isfinite(box_lengths)) or np.any( 

261 box_lengths <= 2 * contact_threshold 

262 ): 

263 # Non-crystallographic entries can carry a placeholder 1 Å box plus 

264 # assembly operators (e.g. EM entry 6hbg). A periodic search radius 

265 # spanning half a box or more is physically ambiguous and makes the 

266 # CellList repeat a huge number of meaningless images. 

267 log.warning( 

268 "Skipping symmetry-mate contacts for invalid/too-small unit cell %s", 

269 box_lengths.tolist(), 

270 ) 

271 return {} 

272 

273 # Get ASU to determine atoms per symmetry copy 

274 asu = get_structure_with_altloc(cif_file, model=1, use_author_fields=False) 

275 asu = asu[~struc.filter_solvent(asu)] 

276 asu = asu[filter_heavy(asu)] 

277 n_asu = len(asu) 

278 n_total = len(unit_cell) 

279 if n_total == n_asu: 279 ↛ 280line 279 didn't jump to line 280 because the condition on line 279 was never true

280 return {} 

281 n_copies = n_total // n_asu 

282 

283 # Label each atom with its symmetry image index 

284 image_idx = np.repeat(np.arange(n_copies), n_asu) 

285 

286 # Use periodic CellList to find contacts across unit cell boundaries 

287 cell_list = struc.CellList( 

288 unit_cell, 

289 cell_size=contact_threshold, 

290 periodic=True, 

291 box=unit_cell.box, 

292 ) 

293 

294 results: dict[tuple[str, int], dict[tuple[str, int], dict[int, set[int]]]] = ( 

295 defaultdict(lambda: defaultdict(lambda: defaultdict(set))) 

296 ) 

297 

298 # For each atom in the ASU (image 0), find contacts with symmetry mates 

299 for i in range(n_asu): 

300 neighbors = cell_list.get_atoms(unit_cell.coord[i], radius=contact_threshold) 

301 for j in neighbors: 

302 if j == i or image_idx[j] == 0: 

303 continue 

304 c1 = ( 

305 unit_cell.label_asym_id[i] 

306 if hasattr(unit_cell, "label_asym_id") 

307 else unit_cell.chain_id[i] 

308 ) 

309 c2 = ( 

310 unit_cell.label_asym_id[j] 

311 if hasattr(unit_cell, "label_asym_id") 

312 else unit_cell.chain_id[j] 

313 ) 

314 r1 = int(unit_cell.res_id[i]) if unit_cell.res_id[i] else 1 

315 r2 = int(unit_cell.res_id[j]) if unit_cell.res_id[j] else 1 

316 atom_serial = i + 1 

317 results[(c1, r1)][(c2, r2)][atom_serial].add(int(image_idx[j])) 

318 

319 return results

362def extract_ligand_links_to_neighbouring_chains( 

363 all_covalent_dict: dict[str, list[tuple[str, str]]], 

364 ligand_asym_id: str, 

365 neighboring_asym_ids: set[str], 

366 link_type: str = "covale", 

367) -> set[str]: 

368 """ 

369 Parse covalant dictionary for a given ligand and its neighbours. 

370 

371 Parameters 

372 ---------- 

373 all_covalent_dict : dict[str, list[tuple[str, str]]] 

374 All covalent links as defined by mmcif annotations 

375 ligand_asym_id : str 

376 ligand asymmetric identification string 

377 neighboring_asym_ids : set[str] 

378 set of neighbour asymmetric identification strings 

379 link_type : str, optional 

380 Covalent linkage type in the dictionary. Options include ``"covale"`` 

381 for a covalent linkage, ``"metalc"`` for a dative bond, and 

382 ``"hydrog"`` for a strong hydrogen bond in a nucleic acid. The 

383 default is ``"covale"``. 

384 

385 Returns 

386 ------- 

387 set[str] 

388 Links between the ligand and its neighbours as ``receptor_end__ligand_end`` 

389 

390 Notes 

391 ----- 

392 For the purpose of covalent annotations, we only consider "covale". 

393 """ 

394 covalent_linkages = set() 

395 if link_type in all_covalent_dict: 

396 for link1, link2 in all_covalent_dict[link_type]: 

397 chain1, chain2 = link1.split(":")[2], link2.split(":")[2] 

398 chains = {chain1, chain2} 

399 if len(chains) == 1: 

400 # remove linkages that are to the same chain! 

401 continue 

402 if ligand_asym_id not in chains: 

403 # only if the ligand is involved 

404 continue 

405 # One chain is the ligand, the other a neighbouring (receptor) 

406 # chain; emit as receptor__ligand. Match whole asym ids, which can be 

407 # multi-character (e.g. "AA") once an entry has more than 26 chains. 

408 if chain1 in neighboring_asym_ids and chain2 == ligand_asym_id: 

409 covalent_linkages.add(f"{link1}__{link2}") 

410 elif chain2 in neighboring_asym_ids and chain1 == ligand_asym_id: 

411 covalent_linkages.add(f"{link2}__{link1}") 

412 return covalent_linkages

456def _find_bridged_interactions( 

457 receptor: "struc.AtomArray", 

458 ligand: "struc.AtomArray", 

459 bridge_atoms: "struc.AtomArray", 

460 receptor_pattern: str, 

461 ligand_pattern: str, 

462 distance_scaling: tuple[float, float], 

463) -> list[tuple[NDArray[np.int_], NDArray[np.int_], NDArray[np.int_]]]: 

464 """Find interactions bridged by intermediary atoms (water or metal). 

465 

466 TODO: remove once peppr has ContactMeasurement.find_bridged_interactions. 

467 """ 

468 import biotite.structure.info as info 

469 from peppr.contacts import ContactMeasurement, find_atoms_by_pattern 

470 

471 if bridge_atoms.array_length() == 0: 471 ↛ 472line 471 didn't jump to line 472 because the condition on line 471 was never true

472 return [] 

473 

474 try: 

475 cm = ContactMeasurement(receptor, ligand) 

476 except Exception as e: 

477 log.warning(f"ContactMeasurement setup failed: {e}") 

478 return [] 

479 

480 receptor_matched = find_atoms_by_pattern(cm._binding_site_mol, receptor_pattern) 

481 ligand_matched = find_atoms_by_pattern(cm._ligand_mol, ligand_pattern) 

482 if len(receptor_matched) == 0 or len(ligand_matched) == 0: 

483 return [] 

484 

485 receptor_coords = cm._binding_site.coord[receptor_matched] 

486 ligand_coords = cm._ligand.coord[ligand_matched] 

487 lo, hi = sorted(distance_scaling) 

488 

489 r_vdw = np.array( 

490 [info.vdw_radius_single(e) for e in cm._binding_site.element[receptor_matched]] 

491 ) 

492 l_vdw = np.array( 

493 [info.vdw_radius_single(e) for e in cm._ligand.element[ligand_matched]] 

494 ) 

495 

496 bridges: list[tuple[NDArray[np.int_], NDArray[np.int_], NDArray[np.int_]]] = [] 

497 for bi in range(bridge_atoms.array_length()): 

498 b_coord = bridge_atoms.coord[bi] 

499 b_vdw = info.vdw_radius_single(bridge_atoms.element[bi]) 

500 

501 r_dists = np.linalg.norm(receptor_coords - b_coord, axis=1) 

502 r_thresholds = r_vdw + b_vdw 

503 r_contacts = receptor_matched[ 

504 (r_dists >= lo * r_thresholds) & (r_dists <= hi * r_thresholds) 

505 ] 

506 if len(r_contacts) == 0: 

507 continue 

508 

509 l_dists = np.linalg.norm(ligand_coords - b_coord, axis=1) 

510 l_thresholds = l_vdw + b_vdw 

511 l_contacts = ligand_matched[ 

512 (l_dists >= lo * l_thresholds) & (l_dists <= hi * l_thresholds) 

513 ] 

514 if len(l_contacts) == 0: 

515 continue 

516 

517 for ri in r_contacts: 

518 for li in l_contacts: 

519 bridges.append( 

520 ( 

521 cm._binding_site_indices[ri : ri + 1], 

522 np.array([li], dtype=int), 

523 np.array([bi], dtype=int), 

524 ) 

525 ) 

526 

527 return bridges

573 ) 

574 ligand_matched = find_atoms_by_pattern(cm._ligand_mol, _METAL_ACCEPTOR_PATTERN) 

575 if len(receptor_matched) == 0 or len(ligand_matched) == 0: 

576 return [] 

577 

578 bridges: list[tuple[NDArray[np.int_], NDArray[np.int_], NDArray[np.int_]]] = [] 

579 for bi in range(coord_metals.array_length()): 

580 b_coord = coord_metals.coord[bi] 

581 r_dists = np.linalg.norm( 

582 cm._binding_site.coord[receptor_matched] - b_coord, axis=1 

583 ) 

584 r_contacts = receptor_matched[r_dists < cutoff] 

585 if len(r_contacts) == 0: 

586 continue 

587 l_dists = np.linalg.norm(cm._ligand.coord[ligand_matched] - b_coord, axis=1) 

588 l_contacts = ligand_matched[l_dists < cutoff] 

589 if len(l_contacts) == 0: 

590 continue 

591 for ri in r_contacts: 

592 for li in l_contacts: 

593 bridges.append( 

594 ( 

595 cm._binding_site_indices[ri : ri + 1], 

596 np.array([li], dtype=int), 

597 np.array([bi], dtype=int), 

598 ) 

599 ) 

600 return bridges

605PEPPR_INTERACTION_TYPES = ( 

606 "hydrogen_bonds", 

607 "salt_bridges", 

608 "pi_stacks",

610 "halogen_bonds",

616def run_peppr_interactions( 

617 receptor: struc.AtomArray, 

618 ligand: struc.AtomArray, 

619 waters: struc.AtomArray, 

620 metals: struc.AtomArray, 

621 ligand_chain: str, 

622) -> tuple[dict[str, dict[int, list[str]]], set[tuple[str, int]], list[str]]: 

623 """Compute interaction hash using peppr ContactMeasurement. 

624 

625 Parameters 

626 ---------- 

627 receptor : AtomArray 

628 Receptor heavy atoms. 

629 ligand : AtomArray 

630 Ligand heavy atoms. 

631 waters : AtomArray 

632 Water heavy atoms. 

633 metals : AtomArray 

634 Metal ion heavy atoms (used for metal bridge detection). 

635 ligand_chain : str 

636 Ligand chain identifier ({instance}.{chain}). 

637 

638 Returns 

639 ------- 

640 interaction_hashes : dict 

641 {instance.chain: {residue_number: [interaction_strings]}} 

642 water_set : set 

643 {(instance.chain, residue_number)} of bridging waters. 

644 failed_interaction_types : list[str] 

645 Interaction types (from :data:`PEPPR_INTERACTION_TYPES`) whose peppr 

646 detector raised, so an empty result for them means "not computed", 

647 not "none found". A failure to build the complex records all types. 

648 """ 

649 interaction_hashes: dict[str, dict[int, list[str]]] = {} 

650 water_set: set[tuple[str, int]] = set() 

651 failed_interaction_types: list[str] = [] 

652 

653 try: 

654 cm = ContactMeasurement(receptor, ligand) 

655 except Exception as e: 

656 components = sorted({str(name) for name in ligand.res_name}) 

657 log.warning( 

658 "run_peppr_interactions: peppr could not build the ligand %s + " 

659 "binding-site complex for contact analysis; skipping its " 

660 "interactions. Components: %s. Underlying error: %s", 

661 ligand_chain, 

662 components, 

663 e, 

664 ) 

665 # No detector could run — every type is unknown, not absent. 

666 return interaction_hashes, water_set, list(PEPPR_INTERACTION_TYPES) 

667 

668 def _add(chain: str, resnr: int, attr: str) -> None: 

669 if chain == ligand_chain: 

670 return 

671 if chain not in interaction_hashes: 

672 interaction_hashes[chain] = {} 

673 if resnr not in interaction_hashes[chain]: 

674 interaction_hashes[chain][resnr] = [] 

675 interaction_hashes[chain][resnr].append(attr) 

676 

677 _PROTEIN_MAINCHAIN = {"N", "CA", "C", "O"} 

678 _NA_MAINCHAIN = {"P", "O5'", "C5'", "C4'", "C3'", "O3'"} 

679 _mainchain_mask = ( 

680 np.isin(receptor.atom_name, list(_PROTEIN_MAINCHAIN)) 

681 & struc.filter_amino_acids(receptor) 

682 ) | ( 

683 np.isin(receptor.atom_name, list(_NA_MAINCHAIN)) 

684 & struc.filter_nucleotides(receptor) 

685 ) 

686 

687 def _is_sidechain(atom_idx: int) -> bool: 

688 return not _mainchain_mask[atom_idx] 

689 

690 # H-bonds 

691 try: 

692 rec_donates, lig_donates = cm.find_hbonds() 

693 for ri, _li in rec_donates: 

694 c = str(receptor.chain_id[ri]) 

695 sc = _is_sidechain(ri) 

696 _add( 

697 c, 

698 int(receptor.res_id[ri]), 

699 f"type:hydrogen_bonds__protisdon:True__sidechain:{sc}", 

700 ) 

701 for ri, _li in lig_donates: 

702 c = str(receptor.chain_id[ri]) 

703 sc = _is_sidechain(ri) 

704 _add( 

705 c, 

706 int(receptor.res_id[ri]), 

707 f"type:hydrogen_bonds__protisdon:False__sidechain:{sc}", 

708 ) 

709 except Exception as e: 

710 log.warning(f"run_peppr_interactions: find_hbonds failed: {e}") 

711 failed_interaction_types.append("hydrogen_bonds") 

712 

713 # Salt bridges 

714 try: 

715 salt_bridges = cm.find_salt_bridges() 

716 for ri, _li in salt_bridges: 

717 c = str(receptor.chain_id[ri]) 

718 _add(c, int(receptor.res_id[ri]), "type:salt_bridges__protispos:True") 

719 except Exception as e: 

720 log.warning(f"run_peppr_interactions: find_salt_bridges failed: {e}") 

721 failed_interaction_types.append("salt_bridges") 

722 

723 # Pi-stacking (deduplicate per residue) 

724 try: 

725 from biotite.structure import PiStacking 

726 

727 stacking = cm.find_stacking_interactions() 

728 seen_stacking: set[tuple[str, int, str]] = set() 

729 for rec_idx, _lig_idx, stack_type in stacking: 

730 ri = rec_idx[0] 

731 c = str(receptor.chain_id[ri]) 

732 stype = "T" if stack_type == PiStacking.PERPENDICULAR else "P" 

733 key = (c, int(receptor.res_id[ri]), stype) 

734 if key not in seen_stacking: 

735 seen_stacking.add(key) 

736 _add(c, int(receptor.res_id[ri]), f"type:pi_stacks__stack_type:{stype}") 

737 except Exception as e: 

738 log.warning(f"run_peppr_interactions: find_stacking_interactions failed: {e}") 

739 failed_interaction_types.append("pi_stacks") 

740 

741 # Pi-cation 

742 try: 

743 pi_cation = cm.find_pi_cation_interactions() 

744 for rec_idx, _lig_idx, cation_in_receptor in pi_cation: 

745 ri = rec_idx[0] 

746 c = str(receptor.chain_id[ri]) 

747 if cation_in_receptor: 

748 _add( 

749 c, 

750 int(receptor.res_id[ri]), 

751 "type:pi_cation__lig_group:Aromatic__protcharged:True", 

752 ) 

753 else: 

754 _add( 

755 c, 

756 int(receptor.res_id[ri]), 

757 "type:pi_cation__lig_group:Cation__protcharged:False", 

758 ) 

759 except Exception as e: 

760 log.warning(f"run_peppr_interactions: find_pi_cation_interactions failed: {e}") 

761 failed_interaction_types.append("pi_cation") 

762 

763 # Halogen bonds 

764 try: 

765 from peppr.common import ( 

766 ACCEPTOR_PATTERN, 

767 HALOGEN_DISTANCE_SCALING, 

768 HALOGEN_PATTERN, 

769 ) 

770 

771 halogen_bonds = cm.find_contacts_by_pattern( 

772 ACCEPTOR_PATTERN, 

773 HALOGEN_PATTERN, 

774 HALOGEN_DISTANCE_SCALING, 

775 ) 

776 for ri, _li in halogen_bonds: 

777 c = str(receptor.chain_id[ri]) 

778 sc = _is_sidechain(ri) 

779 _add(c, int(receptor.res_id[ri]), f"type:halogen_bonds__sidechain:{sc}") 

780 except Exception as e: 

781 log.warning(f"run_peppr_interactions: halogen_bonds failed: {e}") 

782 failed_interaction_types.append("halogen_bonds") 

783 

784 # Water bridges (via plinder patch — peppr public doesn't have this yet) 

785 try: 

786 if waters.array_length() > 0: 

787 from peppr.common import DONOR_PATTERN 

788 from peppr.contacts import find_atoms_by_pattern 

789 

790 receptor_donors = set( 

791 find_atoms_by_pattern(cm._binding_site_mol, DONOR_PATTERN) 

792 ) 

793 w_bridges = find_water_bridges(receptor, ligand, waters) 

794 for rec_idx, _lig_idx, water_idx in w_bridges: 

795 ri = rec_idx[0] 

796 wi = water_idx[0] 

797 # Check if receptor atom is a donor 

798 bs_idx = None 

799 for j, orig_idx in enumerate(cm._binding_site_indices): 799 ↛ 803line 799 didn't jump to line 803 because the loop on line 799 didn't complete

800 if orig_idx == ri: 

801 bs_idx = j 

802 break 

803 protisdon = bs_idx in receptor_donors if bs_idx is not None else True 

804 c = str(receptor.chain_id[ri]) 

805 _add( 

806 c, 

807 int(receptor.res_id[ri]), 

808 f"type:water_bridges__protisdon:{protisdon}", 

809 ) 

810 w_chain = str(waters.chain_id[wi]) 

811 water_set.add((w_chain, int(waters.res_id[wi]))) 

812 except Exception as e: 

813 log.warning(f"run_peppr_interactions: find_water_bridges failed: {e}") 

814 failed_interaction_types.append("water_bridges") 

815 

816 # Metal bridges (via plinder patch — peppr public doesn't have this yet) 

817 try: 

818 if metals.array_length() > 0: 

819 m_bridges = find_metal_bridges(receptor, ligand, metals) 

820 for rec_idx, _lig_idx, metal_idx in m_bridges: 

821 ri = rec_idx[0] 

822 mi = metal_idx[0] 

823 c = str(receptor.chain_id[ri]) 

824 metal_elem = metals.element[mi] 

825 _add( 

826 c, 

827 int(receptor.res_id[ri]), 

828 f"type:metal_complexes__metal_type:{metal_elem}", 

829 ) 

830 except Exception as e: 

831 log.warning(f"run_peppr_interactions: find_metal_bridges failed: {e}") 

832 failed_interaction_types.append("metal_complexes") 

833 

834 return interaction_hashes, water_set, failed_interaction_types
